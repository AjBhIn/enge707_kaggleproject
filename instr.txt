# Git process
# If using branch method
git checkout -b <branch_name>
git add .
git commit -m "<commit_message>"
git push origin <branch_name>

# then merge it
git checkout main
git merge <branch_name>
git push origin main 

# then delete it
git branch -d <branch_name>
